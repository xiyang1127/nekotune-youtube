cmake_minimum_required(VERSION 3.28)
project(NekoTuneYouTube VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Network)
find_package(NekoTuneSDK 3 QUIET)
set(NEKOTUNE_SDK_SOURCE_DIR "" CACHE PATH "NekoTune SDK 3 source directory")
if(NOT NekoTuneSDK_FOUND)
    if(NOT EXISTS "${NEKOTUNE_SDK_SOURCE_DIR}/include/nekotune/sdk/providers.h")
        message(FATAL_ERROR "Set CMAKE_PREFIX_PATH to an installed SDK 3, or NEKOTUNE_SDK_SOURCE_DIR to its source directory")
    endif()
    add_subdirectory("${NEKOTUNE_SDK_SOURCE_DIR}" nekotune-sdk)
endif()
add_library(youtube_core STATIC src/catalog.cpp src/extractor.cpp src/video_proxy.cpp src/extension.cpp)
set_target_properties(youtube_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
target_include_directories(youtube_core PUBLIC src)
target_link_libraries(youtube_core PUBLIC NekoTune::SDK Qt6::Network)
nekotune_add_extension(nekotune_youtube MANIFEST extension.json SOURCES src/plugin.cpp
    FILES qml/Page.qml qml/Controller.qml qml/SettingsPage.qml qml/VideoPlayer.qml
          qml/ActionButton.qml qml/WatchPanel.qml README.md)
target_link_libraries(nekotune_youtube PRIVATE youtube_core)
add_custom_target(extension_package ALL DEPENDS nekotune_youtube_package)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test QuickTest QuickControls2 Multimedia)
    add_executable(fake_extractor tests/fake_extractor.cpp)
    target_link_libraries(fake_extractor PRIVATE Qt6::Core)
    add_executable(youtube_media_smoke tests/media_smoke.cpp)
    target_link_libraries(youtube_media_smoke PRIVATE youtube_core Qt6::Multimedia Qt6::Gui)
    add_executable(youtube_test tests/youtube_test.cpp)
    target_link_libraries(youtube_test PRIVATE youtube_core Qt6::Test)
    target_compile_definitions(youtube_test PRIVATE YOUTUBE_PLUGIN_PATH="$<TARGET_FILE:nekotune_youtube>")
    add_dependencies(youtube_test nekotune_youtube fake_extractor)
    add_test(NAME youtube_core COMMAND youtube_test)
    add_executable(youtube_quick_test tests/quick_test.cpp)
    target_link_libraries(youtube_quick_test PRIVATE Qt6::QuickTest Qt6::QuickControls2 Qt6::Multimedia)
    target_compile_definitions(youtube_quick_test PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/qml")
    add_test(NAME youtube_interface COMMAND youtube_quick_test)
    set_tests_properties(youtube_interface PROPERTIES ENVIRONMENT
        "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QUICK_CONTROLS_STYLE=Fusion;QML2_IMPORT_PATH=${CMAKE_CURRENT_SOURCE_DIR}/tests/imports")
endif()
